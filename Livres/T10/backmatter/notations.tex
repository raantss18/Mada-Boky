\chapter*{Notations utilis\'ees}
\addcontentsline{toc}{chapter}{Notations utilis\'ees}

\begin{multicols}{2}
\begin{description}[leftmargin=2.5cm, font=\color{BN}\bfseries]
  \item[$\N,\Z,\Q,\R$] Ensembles de nombres
  \item[$\abs{x}$] Valeur absolue de $x$
  \item[$\norme{\vec u}$] Norme du vecteur $\vec u$
  \item[$\vv{AB}$] Vecteur de $A$ vers $B$
  \item[$\vec u\cdot\vec v$] Produit scalaire
  \item[$\lnot P$] N\'egation de $P$
  \item[$P\land Q$] $P$ et $Q$
  \item[$P\lor Q$] $P$ ou $Q$ (inclusif)
  \item[$P\Rightarrow Q$] $P$ implique $Q$
  \item[$P\Leftrightarrow Q$] $P$ \'equivaut \`a $Q$
  \item[$\forall$] Pour tout
  \item[$\exists$] Il existe
  \item[$\Delta$] Discriminant ($b^2-4ac$)
  \item[$\pgcd(a,b)$] Plus grand commun diviseur
  \item[$a\mid b$] $a$ divise $b$
  \item[$a\equiv b\pmod n$] $a$ congru \`a $b$ modulo $n$
\end{description}
\end{multicols}
